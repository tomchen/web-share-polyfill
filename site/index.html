          Usage

          Install

          npm i web-share-polyfill

          import 'web-share-polyfill/common'

button.addEventListener('click', () => {
  navigator.share({ title: document.title, url: location.href }).catch(() => {})
})

          import { polyfill } from 'web-share-polyfill'
import { copy, qr, email, whatsapp, facebook, x, telegram } from 'web-share-polyfill/targets'
import { fr, zh, zhHant, ja } from 'web-share-polyfill/locales'

polyfill({
  targets: [copy, qr, email, whatsapp, facebook, x, telegram],
  locales: [fr, zh, zhHant, ja], // English is built in
})

          import { share } from 'web-share-polyfill'
import { copy, qr, x } from 'web-share-polyfill/targets'

const id = await share({ url: location.href }, { targets: [copy, qr, x] })
// 'native', 'copy', 'qr' or 'x'

          <script src="https://cdn.jsdelivr.net/npm/web-share-polyfill"></script>
<script>
  WebSharePolyfill.polyfill({ targets: { '*': ['copy', 'qr', 'x'], zh: ['copy', 'qr', 'wechat', 'weibo'] } })
</script>
